`timescale 1ns / 1ps

module ALU_TB;

reg [31:0] A;
reg [31:0] B;
reg [3:0] ALU_Sel;

wire [31:0] ALU_Out;
wire Zero;

ALU01 uut (
    .A(A),
    .B(B),
    .ALU_Sel(ALU_Sel),
    .ALU_Out(ALU_Out),
    .Zero(Zero)
);

initial begin

    // AND
    A = 12;
    B = 10;
    ALU_Sel = 4'b0000;
    #10;

    // OR
    A = 12;
    B = 10;
    ALU_Sel = 4'b0001;
    #10;

    // ADD
    A = 25;
    B = 15;
    ALU_Sel = 4'b0010;
    #10;

    // SUB
    A = 50;
    B = 20;
    ALU_Sel = 4'b0110;
    #10;

    // XOR
    A = 9;
    B = 5;
    ALU_Sel = 4'b0011;
    #10;

    // SLT
    A = 5;
    B = 10;
    ALU_Sel = 4'b0111;
    #10;

    // Zero flag
    A = 10;
    B = 10;
    ALU_Sel = 4'b0110;
    #10;

    $finish;

end

endmodule
